<script>

function analyzeFRP() {

    const form = new FormData(document.getElementById('frpDeepForm'));

    const f = parseInt(form.get('f_val'));

    const r = parseInt(form.get('r_val'));

    const p = parseInt(form.get('p_val'));

    

    const res = document.getElementById('frpDeepResult');

    res.style.display = 'block';

    

    let lead = "";

    let advice = "";

    let color = "";


    if (f >= r && f >= p) {

        lead = "Primary Root: FEAR";

        color = "#e74c3c";

        advice = "<strong>The Catastrophe Lens:</strong> Your control is an attempt to prevent pain. You see your partner as 'unsafe' or 'unreliable.' <br><br><strong>Repair Action:</strong> Practice 'Surrender Sessions.' Let your partner manage a high-stakes task (like finances or travel) without your input. Observe your anxiety without acting on it.";

    } else if (r >= f && r >= p) {

        lead = "Primary Root: RIGIDITY";

        color = "#3498db";

        advice = "<strong>The Perfection Lens:</strong> You value 'The Way' more than the person. You treat your relationship like an operations manual.<br><br><strong>Repair Action:</strong> The 'Gray Area' Exercise. Purposely do something 'the wrong way' once a day. Learn that the relationship survives imperfection.";

    } else {

        lead = "Primary Root: PRIDE";

        color = "#27ae60";

        advice = "<strong>The Pedestal Lens:</strong> You view yourself as the 'Parent' and your partner as the 'Child.' This kills respect and intimacy.<br><br><strong>Repair Action:</strong> The 'Equality Audit.' Ask your partner for their opinion on three things today and <em>accept their influence</em> completely, even if you disagree.";

    }


    res.innerHTML = "<h3 style='color:"+color+"; margin-top:0;'>"+lead+"</h3>" + advice;

}

</script>
